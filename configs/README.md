## Назначение
Блое /configs предназначен для хранения конфигурационных файлов, которые используются для задания начальных параметров и настроек модели, включая характеристики сетки, физические свойства среды

## Структура файла
рассмотрим структуру файла поподробнее

`````
 "simulation": {  //параметры для настройки симуляции
        "Lx": 2.0,  /длина области симуляции по оси х
        "Ly": 1.0,  //ширина области симуляции по оси у
        "nx": 128,  //количество точек по оси х, определяет разрешающую способность сетки
        "ny": 64,  //количество точек по оси у
        "T": 4.0,  //полное время симуляции
        "dt": 0.001,  //шат по времени, с которым выполняются вычисления
        "D": 0.01,
        "U_max": 1.0,  //максимальное значение скорости течения
        "save_every": 100  //переодичность записи результатов
    },
    "model": {  //параметры модели
        "modes1": 16,
        "modes2": 16,
        "width": 64,
        "n_layers": 4
    },
    "training": {  //параметры обучения нейронной сети
        "n_samples": 500,  //кол-во выборок для обучения
        "batch_size": 8,  //размер батчей данных, обрабатываемых за одну итерацию
        "lr": 0.001,  //скорость обучения
        "weight_decay": 0.0001,  //регуляризация
        "n_epochs": 100,  //количество эпох
        "train_ratio": 0.8  //доля данных для обучения
    },
    "solver_generation": {
        "seed": 42,
        "grid": {  //блок, описывающей сетку
            "L": 60.0,  //длина области расчёта
            "H": 60.0,  //ширина области расчёта
            "Nx": 100,  //количество узлов сетки по направлению оси х
            "Ny": 100,  //количество узлов сетки по направлению оси у
            "Tmax": 400.0   //максимальное время моделирования
        },
        "params": {  //диапазон и шаг генерируемых параметров
            "c_in": {  //концентрация проппанта
                "min": 0.05,  //минимальное значение
                "max": 0.45,  //максимальное значение
                "step": 0.05  //шаг изменения
            },
      "physics": {  //набор физических констант
            "beta": 2.5,  //эксперементальная зависимость вязкости от концентрации проппанта
            "cmax": 0.635,  //максимальная концентрация проппанта
            "rho1": 1000.0,  //плотность жидкости
            "rho2": 2700.0,  //плотность проппанта
            "g": 0.0,  //учет гравитации
            "r": 0.0  //радиус частиц проппанта
        },
`````
